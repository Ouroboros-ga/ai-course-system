## 🤖 智能体教学功能设计

### 1. 教学智能体架构

```python
class TeachingAgent:
    """
    教学智能体: 自适应数据结构与算法教学
    """
    
    def __init__(self):
        self.state = "idle"  # idle, teaching, assessing, reflecting
        self.knowledge_graph = ComputerKnowledgeGraph()
        self.student_model = StudentModel()
        self.llm = ComputerBrainLLM()
        self.visualizer = AlgorithmVisualizer()
        
    def perceive(self, student_action):
        """
        感知模块: 分析学生操作和理解程度
        
        输入:
            student_action: {
                "action_type": "edit_node" | "run_algorithm" | "ask_question",
                "action_data": {...},
                "timestamp": "2026-06-21T10:00:00"
            }
        
        输出:
            StudentState: {
                "understanding_level": 0.0-1.0,  # 理解程度评分
                "knowledge_gaps": ["链表插入", "时间复杂度分析"],  # 知识缺口
                "engagement_level": "high" | "medium" | "low",  # 参与度
                "preferred_learning_style": "visual" | "textual" | "interactive"
            }
        """
        # 1. 分析学生操作数据
        operation_analysis = self.analyze_operation(student_action)
        
        # 2. 评估理解程度（基于操作正确性、问题质量等）
        understanding_score = self.evaluate_understanding(operation_analysis)
        
        # 3. 检测知识缺口（基于错误模式、问题内容等）
        knowledge_gaps = self.detect_knowledge_gaps(operation_analysis)
        
        # 4. 判断参与度（基于操作频率、停留时间等）
        engagement_level = self.assess_engagement(student_action)
        
        return StudentState(
            understanding_level=understanding_score,
            knowledge_gaps=knowledge_gaps,
            engagement_level=engagement_level,
            preferred_learning_style=self.student_model.learning_style
        )
    
    def reason(self, student_state):
        """
        推理模块: 决策教学策略和下一步动作
        
        输入:
            student_state: StudentState对象
        
        输出:
            TeachingStrategy: {
                "strategy_type": "explain_concept" | "visualize_demo" | "interactive_practice",
                "content_topic": "链表插入操作",
                "visualization_type": "animation" | "interactive_edit",
                "difficulty_level": "basic" | "intermediate" | "advanced",
                "next_steps": ["讲解原理", "演示动画", "让学生操作"]
            }
        """
        # 1. 根据理解程度决定教学策略
        if student_state.understanding_level < 0.3:
            # 理解程度低：回归基础概念讲解
            strategy = self.generate_basic_concept_strategy(student_state)
        elif student_state.understanding_level < 0.6:
            # 理解程度中等：可视化演示+交互练习
            strategy = self.generate_visualization_strategy(student_state)
        elif student_state.understanding_level < 0.8:
            # 理解程度较高：复杂场景应用
            strategy = self.generate_advanced_application_strategy(student_state)
        else:
            # 理解程度高：拓展新知识点
            strategy = self.generate_next_topic_strategy(student_state)
        
        # 2. 根据知识缺口调整教学内容
        if student_state.knowledge_gaps:
            strategy.content_topic = student_state.knowledge_gaps[0]
        
        # 3. 根据学习风格调整呈现方式
        if student_state.preferred_learning_style == "visual":
            strategy.visualization_type = "animation"
        elif student_state.preferred_learning_style == "interactive":
            strategy.visualization_type = "interactive_edit"
        
        return strategy
    
    def execute(self, teaching_strategy):
        """
        执行模块: 生成教学内容和可视化指令
        
        输入:
            teaching_strategy: TeachingStrategy对象
        
        输出:
            TeachingContent: {
                "explanation_text": "链表插入操作分为三步...",
                "visualization_commands": [
                    {"type": "highlight_node", "node_id": 3},
                    {"type": "animate_insertion", "position": "after_node_2"}
                ],
                "interactive_prompts": ["请尝试在节点2后插入新节点"],
                "code_example": "def insert_node(head, data): ...",
                "quiz_questions": [...]
            }
        """
        # 1. 生成讲解文本（调用垂类大模型）
        explanation_text = self.llm.generate_explanation(
            topic=teaching_strategy.content_topic,
            difficulty=teaching_strategy.difficulty_level
        )
        
        # 2. 生成可视化指令
        visualization_commands = self.visualizer.generate_commands(
            topic=teaching_strategy.content_topic,
            visualization_type=teaching_strategy.visualization_type
        )
        
        # 3. 生成交互提示
        interactive_prompts = self.generate_interactive_prompts(teaching_strategy)
        
        # 4. 生成代码示例

            teaching_result: {

                "correctness_rate": 0.8,
                "time_spent": 120,
                "student_feedback": "讲解很清晰"

                "effectiveness_score": 0.85,

            }

        self.student_model.update(teaching_result)

        self.knowledge_graph.update_learning_path(teaching_result)
```

```python
# 计算机学科知识库构建
class ComputerKnowledgeBase:
    def __init__(self):
        self.graph_db = Neo4jConnection()  # 知识图谱
        self.vector_db = MilvusConnection()  # 向量检索
        self.llm = ComputerBrainLLM()
        
    def build_knowledge_graph(self):
        """构建计算机学科知识图谱"""
        # 1. 提取实体（数据结构、算法、概念）
        entities = self.extract_entities_from_textbooks()
        
        # 2. 提取关系（依赖、包含、应用）
        relations = self.extract_relations_from_textbooks()
        
        # 3. 存储到Neo4j
        for entity in entities:
            self.graph_db.create_node(entity)
        
        for relation in relations:
            self.graph_db.create_relation(relation)
    
    def build_vector_index(self):
        """构建向量检索索引"""
        # 1. 收集文档（教材、代码、论文）
        documents = self.collect_documents()
        
        # 2. 文档分块
        chunks = self.split_documents(documents)
        
        # 3. 向量化
        embeddings = self.llm.embed(chunks)
        
        # 4. 存储到Milvus
        self.vector_db.insert(embeddings, chunks)
    
    def retrieve_knowledge(self, query):
        """检索相关知识"""
        # 1. 向量检索
        vector_results = self.vector_db.search(query)
        
        # 2. 知识图谱检索
        graph_results = self.graph_db.search(query)
        
        # 3. 合并结果
        return self.merge_results(vector_results, graph_results)
```
